rocket_launch 快速开始

几分钟内即可上手

Bash / Shell 
# 步骤 1: 安装依赖
# 更新包管理器
pkg update && pkg upgrade # 适用于 Termux
# 或者
sudo apt update && sudo apt upgrade # 适用于 Linux

# 如果未安装 Python，则安装
pkg install python # Termux
# 或者
sudo apt install python3 # Linux

# 安装所需的 Python 包
pip install requests colorama

# 步骤 2: 克隆/创建 ZynoxAI
# 下载脚本
git clone https://github.com/BugeStudioTeam/Zynox.git

# 重命名
mv Zynox ZynoxAI
cd ~/ZynoxAI

# 步骤 3: 设置别名（可选但推荐）
# 将别名添加到 ~/.bashrc
echo "alias zynox='python ~/ZynoxAI/zynox.py'" >> ~/.bashrc
source ~/.bashrc

# 或者对于 Termux 使用 ~/bin 目录
mkdir -p ~/bin
ln -s ~/ZynoxAI/zynox.py ~/bin/zynox
chmod +x ~/ZynoxAI/zynox.py

# 步骤 4: 验证安装
# 应该显示 ZYNOX 标志
zynox

# 应该显示帮助菜单
zynox --help

# 应该显示关于菜单
zynox --about
创建文件夹 
zynox "创建一个名为 myproject 的文件夹"
搜索并读取文件 
zynox "找到 abc.txt 并读取它"
执行系统命令 
zynox "列出所有文件"
启动 Web 服务器（端口 5000） 
zynox --web
启动 Telegram 机器人 
zynox --telegram-bot TOKEN
开始新的对话 
zynox --new-session
